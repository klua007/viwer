4. Aplicar a correção no SEU bot

    

      O sintoma "carrega e tenta puxar a sessão anterior apagada" acontece porque as creds continuam no disco:
      o Baileys vê registered=true e tenta reusar um aparelho que você já removeu no celular → o servidor
      devolve 401 loggedOut e o bot entra em loop. O trecho abaixo apaga sozinho e pede um QR novo.
    

    

      Comandos no terminal 

# 1) PARE todos os processos (2 processos na mesma sessão = erro 440)
pm2 stop all            # ou: pkill -f node

# 2) Descubra a pasta certa (o erro nº 1 é apagar a pasta errada)
npm run doctor -- --dir ./session

# 3) Apague com o processo PARADO
rm -rf ./session        # pasta do useMultiFileAuthState
rm -f  ./store.json     # só se usar makeInMemoryStore

# 4) Recrie a sessão
npm run pair                              # QR no terminal
npm run pair -- --phone 5519999999999     # ou por código de pareamento
npm start                                 # ou pelo painel web

    

    

      Código de reconexão com auto-limpeza 

import { makeWASocket, useMultiFileAuthState, DisconnectReason,
         fetchLatestBaileysVersion, Browsers } from '@boruto_vk7/baileys'
import { Boom } from '@hapi/boom'
import { rm } from 'node:fs/promises'

const SESSION_DIR = './session'        // ⚠️ exatamente a pasta que o seu bot usa
const PHONE = ''                       // opcional: '5519999999999' p/ código de pareamento

async function start(fresh = false) {
  // fresh = true → sessão nova do zero (é o que falta no seu caso)
  if (fresh) await rm(SESSION_DIR, { recursive: true, force: true })

  const { version } = await fetchLatestBaileysVersion()
  const { state, saveCreds } = await useMultiFileAuthState(SESSION_DIR)

  const sock = makeWASocket({
    version,
    auth: state,
    printQRInTerminal: true,           // QR no terminal
    browser: Browsers.ubuntu('Meu Bot'),
    qrTimeout: 45_000
  })

  sock.ev.on('creds.update', saveCreds)

  sock.ev.on('connection.update', async ({ connection, lastDisconnect, qr }) => {
    if (qr) console.log('🔳 QR gerado — escaneie no celular')

    // pareamento por número (sem QR), se você preencheu PHONE
    if (connection === 'connecting' && !sock.authState.creds.registered && PHONE) {
      setTimeout(async () => {
        console.log('🔢 Código:', await sock.requestPairingCode(PHONE))
      }, 1800)
    }

    if (connection === 'close') {
      const code = new Boom(lastDisconnect?.error)?.output?.statusCode

      // 401 loggedOut / 500 badSession = sessão morta → APAGA e re-pareia
      if (code === DisconnectReason.loggedOut || code === DisconnectReason.badSession) {
        console.warn('Sessão inválida (', code, ') — apagando creds e pedindo novo QR')
        return start(true)
      }
      // 440 = outro processo usando a mesma sessão: não adianta reconectar
      if (code === DisconnectReason.connectionReplaced) {
        console.error('Outro processo está com essa sessão. Encerre o duplicado.')
        return
      }
      console.warn('Reconectando… código', code)
      return start(false)
    }

    if (connection === 'open') console.log('✅ conectado como', sock.user?.id)
  })
}

start(false)
